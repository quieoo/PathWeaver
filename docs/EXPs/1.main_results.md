````bash
nohup bash -c '
set -euo pipefail
COMMON_ARGS=(
  --model-path /mnt/n0/models/qwen3-4B-Instruct
  --embedding-model /mnt/n0/models/bge-en-v1.5/
  --n-samples 100
  --similarity-top-k 64
  --max-model-len 65536
  --gpu-memory-utilization 0.8
  --use-lmcache
  --recompute-ratios 0.05
  --lmcache-warmup-mode reuse
)

DATASETS=(
  "/mnt/n0/datasets/wiki_hotspot_musique/merged_data/source_data/mintqa_test_pruned256.jsonl|/mnt/n0/datasets/wiki_hotspot_musique/merged_data/source_data/mintqa_test_pruned64_hop2.jsonl||/mnt/n0/PathWeaver/experiments/vector_rag_index/mintqa_dev_bge_256"
)

for entry in "${DATASETS[@]}"; do
  IFS="|" read -r dataset_path queryset_path dataset_type index_path <<< "$entry"

  cmd=(
    python3 /mnt/n0/PathWeaver/experiments/vector_rag.py
    --dataset-path "$dataset_path"
    --index-path "$index_path"
    "${COMMON_ARGS[@]}"
  )

  if [[ -n "$queryset_path" ]]; then
    cmd+=(--queryset-path "$queryset_path")
  fi

  if [[ -n "$dataset_type" ]]; then
    cmd+=(--dataset-type "$dataset_type")
  fi

  "${cmd[@]}"
done

' >> /mnt/n0/PathWeaver/docs/EXPs/main_results_vector_rag_cacheblend_top64_recomp0.05.log 2>&1 &

````
