# 如果你想从1.0开始，只需要改成例如: MIX_RATIOS=("1.0" "0.7" "0.3" "0.0")
MIX_RATIOS=(
  "0.8"
  "0.6"
  "0.2"
  "0.0"
)

SEED=42

for dataset_stem in "${DATASETS[@]}"; do
  base_json="${DATA_DIR}/${dataset_stem}.jsonl"
  echo "[$(date -Iseconds)] dataset start: ${dataset_stem}"

  for idx in "${!FLIP_RATIOS[@]}"; do
    flip_ratio="${FLIP_RATIOS[$idx]}"
    mix_ratio="${MIX_RATIOS[$idx]}"
    flip_tag="$(printf "%03d" "$(python - <<PY
ratio = float("${flip_ratio}")
print(int(round(ratio * 100)))
PY
")")"

    noisy_stem="${dataset_stem}_flip${flip_tag}"
    noisy_json="${DATA_DIR}/${noisy_stem}.jsonl"
    key_npy="${DATA_DIR}/${noisy_stem}_qwen-embedding-0.6B_embd_key.npy"
    value_npy="${DATA_DIR}/${noisy_stem}_qwen-embedding-0.6B_embd_value.npy"

    echo "[$(date -Iseconds)] start: dataset=${dataset_stem}, flip_ratio=${flip_ratio}, mix_ratio=${mix_ratio}"

    python /mnt/n0/PathWeaver/docs/experiments/sensitivity/create_noise_edge.py \
      --input "${base_json}" \
      --output "${noisy_json}" \
      --flip_ratio "${flip_ratio}" \
      --seed "${SEED}" \
      --progress

    python /mnt/n0/PathWeaver/docs/scripts/embedding_v2.py \
      --model_name qwen3-embedding-0.6B \
      --local_model_path "${EMBED_MODEL_DIR}/" \
      --dataset_type dag \
      --dataset_path "${noisy_json}" \
      --batch_size 1024 \
      --progress

    python3 /mnt/n0/PathWeaver/experiments/eval_generation.py generation \
      --dataset_dir "${DATA_DIR}" \
      --test_dataset "${noisy_stem}.jsonl" \
      --model_dir "${MODEL_DIR}" \
      --encoder_dir "${ENCODER_DIR}" \
      --encoder_spec qwen-embedding-0.6B \
      --llm_base_dir "${LLM_BASE_DIR}" \
      --llm_type qwen3 \
      --dataset_type dag \
      --precomputed_embed_keys_path "${key_npy}" \
      --precomputed_embed_values_path "${value_npy}" \
      --kb_layer_frequency 3 \
      --kb_size 10 \
      --query_size 100 \
      --path_attn \
      --step 7800 \
      --t_step 8000 \
      --kb_scale_factor 4 \
      --path_attn_mix_ratio "${mix_ratio}"

    echo "[$(date -Iseconds)] done:  dataset=${dataset_stem}, flip_ratio=${flip_ratio}, mix_ratio=${mix_ratio}"
  done

  echo "[$(date -Iseconds)] dataset done: ${dataset_stem}"
done
' > /mnt/n0/PathWeaver/experiments/sensitivity_logs/edge_flip_batched.log 2>&1 &
