## MSA Breakdown
````bash
nohup bash -c '

run_case() {
  local numdoc="$1"
  local gpus="$2"

  echo "[$(date -Iseconds)] start: numdoc=${numdoc}, gpus=${gpus}"
  CUDA_VISIBLE_DEVICES="${gpus}" \
  python /mnt/n0/PathWeaver/experiments/msa.py \
    --dataset-path /mnt/n0/datasets/wiki_hotspot_musique/merged_data/source_data/msa_merged_memory_docs.json \
    --queryset-path /mnt/n0/datasets/wiki_hotspot_musique/merged_data/source_data/2wiki_dev_2hop.json \
    --model-path /mnt/n0/models/MSA-4B \
    --dataset-limit "${numdoc}" \
    --memory-cache-dir /mnt/n0/PathWeaver/experiments/msa_cache/msa_merged_memory_${numdoc}docs_cache \
    --n-samples 10 \
    --block-size 2048 \
    --max-batch-size 1 \
    --doc-top-k 64
  echo "[$(date -Iseconds)] done:  numdoc=${numdoc}, gpus=${gpus}"
}

run_case 1000 1
run_case 8000 1
run_case 64000 1,2
run_case 512000 1,2,3
' > /mnt/n0/PathWeaver/experiments/motivation_study/msa_reuse_efficiency.log 2>&1 &
````
